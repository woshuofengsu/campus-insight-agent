### 2.1 新增 `agent/llm_client.py`
```python
import json, time, urllib.request, threading, logging
from config import DEEPSEEK_API_KEY, DEEPSEEK_BASE_URL, DEEPSEEK_MODEL
_log = logging.getLogger(__name__)
_LOCK = threading.Lock(); _FAIL = {"n": 0, "open_until": 0.0}
_FAIL_THRESHOLD, _OPEN_SECONDS, _DEFAULT_TIMEOUT = 3, 60, 6

def chat(messages, *, module, purpose="", temperature=0.2, max_tokens=512,
         timeout=_DEFAULT_TIMEOUT, uid=None):
    """统一 LLM 调用。返回 dict(ok,text,raw,tokens_in,tokens_out,duration_ms,error)。
    - 熔断：连续失败≥阈值后 60s 内直接快速失败（不阻塞请求线程）
    - 任何异常都不抛出，调用方按 ok=False 走规则兜底
    - 成功/失败均写 llm_usage（失败 token=0）"""
    import time as _t
    now = _t.time()
    with _LOCK:
        if now < _FAIL["open_until"]:
            return {"ok": False, "error": "circuit_open", "text": ""}
    if not DEEPSEEK_API_KEY:
        return {"ok": False, "error": "no_key", "text": ""}
    payload = json.dumps({"model": DEEPSEEK_MODEL, "messages": messages,
                          "temperature": temperature, "max_tokens": max_tokens}).encode()
    req = urllib.request.Request(DEEPSEEK_BASE_URL + "/chat/completions", data=payload,
        headers={"Authorization": f"Bearer {DEEPSEEK_API_KEY}", "Content-Type": "application/json"})
    t0 = _t.time()
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            data = json.loads(resp.read().decode())
        dt = int((_t.time()-t0)*1000)
        txt = data["choices"][0]["message"]["content"].strip()
        u = data.get("usage", {}); ti = u.get("prompt_tokens", 0); to = u.get("completion_tokens", 0)
        with _LOCK: _FAIL.update(n=0, open_until=0.0)
        from data.db_llm_usage import record_usage
        try: record_usage(module, ti, to, dt, input_preview=(purpose or "")[:80])
        except Exception: _log.debug("record usage fail", exc_info=True)
        return {"ok": True, "text": txt, "raw": data, "tokens_in": ti, "tokens_out": to, "duration_ms": dt}
    except Exception as e:
        dt = int((_t.time()-t0)*1000)
        with _LOCK:
            _FAIL["n"] += 1
            if _FAIL["n"] >= _FAIL_THRESHOLD: _FAIL["open_until"] = _t.time()+_OPEN_SECONDS
        _log.warning("llm call fail: %s", e)
        return {"ok": False, "error": str(e)[:120], "text": "", "duration_ms": dt}
```

### 3.2 新增 `agent/policy_rag.py`
```python
"""政策 RAG：仅基于检索到的已发布知识片段生成，强制引用，引用不过即弃用转人工。"""
from config import POLICY_LLM_RAG
from agent.llm_client import chat

_SYS = ("你是社区政策问答助手。只能依据【材料】回答，不得使用材料外知识，不得给医疗/法律结论。"
        "材料不足时只输出 INSUFFICIENT。回答末尾必须另起一行，原样引用所用条目标题，格式：参考：<标题>。")

def _context(results):
    blocks=[]
    for i,e in enumerate(results[:5],1):
        body=(e.get("plain_interpretation") or e.get("summary") or e.get("content") or "")[:600]
        blocks.append(f"[{i}] 标题：{e.get('title')}（V{e.get('version') or 1}）\n正文：{body}")
    return "\n\n".join(blocks), {e.get("title") for e in results[:5]}

def generate(question, results):
    if not POLICY_LLM_RAG or not results:
        return None
    ctx, titles = _context(results)
    r = chat([{"role":"system","content":_SYS},
              {"role":"user","content":f"【材料】\n{ctx}\n\n【问题】{question}"}],
             module="policy_rag", purpose=question[:60], temperature=0.1, max_tokens=400, timeout=8)
    if not r["ok"]: return None
    ans = r["text"].strip()
    if "INSUFFICIENT" in ans[:40]: return None
    # 引用一致性：答案必须引用本次检索片段里真实存在的标题（堵住子串伪造）
    if not any(t and t in ans for t in titles): return None
    return {"answer": ans, "cited": next((t for t in titles if t in ans), ""),
            "tokens_in": r["tokens_in"], "tokens_out": r["tokens_out"]}
```

## WS6 加密正名：真 AES-256-GCM + 密文版本前缀 + 平滑轮换

**现状**：`utils/crypto.py` 实为 scrypt + HMAC-CTR 自研流密码（docstring 已诚实声明非 AES）。对外要讲 AES-256-GCM，就把实现补齐，且**不能让现有密文失效**。

### 6.1 依赖
`requirements.txt` 增加 `cryptography>=42`；`py -3.12 -m pip install cryptography`。
### 6.2 改造 `utils/crypto.py`（保持 `encrypt/decrypt` 对外接口不变）
- 新密文加版本前缀：`g1$<base64(nonce|ct|tag)>`（GCM）；旧密文无前缀，走旧 HMAC-CTR 逻辑。
```python
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
import os, base64, hashlib
class Crypto:
    def __init__(self, key_material):
        self._key = hashlib.sha256(key_material.encode()).digest()  # 32B = AES-256
        self._aes = AESGCM(self._key)
        # 保留旧 scrypt/HMAC-CTR 派生，仅用于解密历史密文
        self._legacy = _LegacyStream(key_material)
    def encrypt(self, plaintext):
        nonce = os.urandom(12)
        ct = self._aes.encrypt(nonce, plaintext.encode(), None)
        return "g1$" + base64.b64encode(nonce+ct).decode()
    def decrypt(self, token):
        if token.startswith("g1$"):
            raw = base64.b64decode(token[3:]); nonce, ct = raw[:12], raw[12:]
            return self._aes.decrypt(nonce, ct, None).decode()
        return self._legacy.decrypt(token)     # 旧密文可读
```
- 把现有 HMAC-CTR 实现收进内部 `_LegacyStream`（只用于解密，不再用于新加密）。
### 6.3 存量重加密脚本 `scripts/reencrypt_phones.py`
遍历所有 `*_enc` 列（user_profile.phone_enc、emergency_contacts、community_issues 三个、health_consults、emergency_calls 等，与 v36/v38/v39 覆盖列一致），对非 `g1$` 前缀的密文：`decrypt→encrypt` 回写，**先备份 db 再执行、逐行 try、输出统计**。幂等（已是 g1$ 跳过）。
### 6.4 测试 `tests/test_crypto_aes.py`
